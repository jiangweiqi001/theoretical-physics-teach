```python
def check_lesson_structure(root: Path, note: IndexedNote, issues: list[Issue]) -> None:
    lessons_root = (root / "lessons").resolve()
    try:
        note.path.relative_to(lessons_root)
    except ValueError:
        return

    h2s: list[tuple[int, str]] = []
    for line_number, line in enumerate(note.lines, start=1):
        match = re.match(r"^\s{0,3}##\s+(.+?)\s*$", line)
        if match:
            h2s.append((line_number, match.group(1).rstrip("#").rstrip()))

    if not h2s or h2s[0][1] != "Notation and conventions":
        issues.append(Issue(note.relative_path, h2s[0][0] if h2s else 1, "lesson-first-h2", "first lesson H2 must be exactly 'Notation and conventions'"))
    if not h2s or h2s[-1][1] != "End-of-Note checkpoint":
        issues.append(Issue(note.relative_path, h2s[-1][0] if h2s else 1, "lesson-final-h2", "final lesson H2 must be exactly 'End-of-Note checkpoint'"))
```

Call it for each indexed Note before link checking.
